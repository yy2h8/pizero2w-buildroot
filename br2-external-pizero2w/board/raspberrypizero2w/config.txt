# Raspberry Pi Zero 2W config.txt - Minimal Configuration

# Kernel
kernel=zImage
arm_64bit=0

# Serial Console
enable_uart=1
uart_2ndstage=1
dtoverlay=miniuart-bt

# GPU Memory (minimal allocation)
gpu_mem=16

# Boot Speed Optimizations
boot_delay=2
disable_splash=1

# Power Management (disable unused features)
dtparam=audio=off

# WiFi/SDIO Configuration
# NOTE: The Pi Zero 2W WiFi is built-in and configured in bcm2710-rpi-zero-2-w.dtb
# The 'sdio' overlay is for EXTERNAL SDIO devices on GPIOs 22-27, NOT for built-in WiFi
# Using the overlay was causing mmc1 initialization failures
# Removed: dtoverlay=sdio
# Removed: dtparam=sd_poll_once=off
